% Internal structure of the model, block-wise (bottom to top), in the style of the Transformer diagram.
% Layer sizes: bench.py (Enc, Model, DATASETS) and extract_crema_av.py (ResNet-18 features, log-mel statistics).
\begin{tikzpicture}[
  font=\small, >=Stealth,
  b/.style={draw, thick, rounded corners=3pt, minimum width=34mm, minimum height=6.2mm, inner sep=1.5pt,
            align=center, fill=#1},
  conv/.style={b=orange!30}, pool/.style={b=red!14}, norm/.style={b=yellow!35}, lin/.style={b=cyan!18},
  soft/.style={b=green!22}, act/.style={b=yellow!18},
  box/.style={draw, thick, rounded corners=7pt, fill=gray!9, inner xsep=4mm, inner ysep=2.6mm},
  up/.style={->, thick},
  side/.style={font=\footnotesize, align=center},
  io/.style={font=\footnotesize, align=center}]

\def\g{2.6mm}   % vertical gap between blocks

%════════ video tower (x = 0) ════════
\node[act] (v5) at (0,0) {ReLU};
\node[lin, below=\g of v5] (v4) {Linear $256\to64$};
\node[act, below=\g of v4] (v3) {Dropout $p{=}0.3$};
\node[act, below=\g of v3] (v2) {ReLU};
\node[lin, below=\g of v2] (v1) {Linear $512\to256$};
\node[norm, below=6mm of v1] (vstd) {Standardise};
\node[pool, below=\g of vstd] (vmean) {Mean over 8 frames};
\node[pool, below=6mm of vmean] (vgap) {Global average pool};
% residual block (repeated)
\node[norm, below=6mm of vgap] (radd) {Add \& ReLU};
\node[norm, below=\g of radd] (rbn2) {BatchNorm};
\node[conv, below=\g of rbn2] (rc2) {Conv $3{\times}3$, $C$};
\node[norm, below=\g of rc2] (rbn1) {BatchNorm \& ReLU};
\node[conv, below=\g of rbn1] (rc1) {Conv $3{\times}3$, $C$};
\node[pool, below=7mm of rc1] (vmp) {Max pool $3{\times}3$, /2};
\node[conv, below=\g of vmp] (vstem) {Conv $7{\times}7$, 64, /2 \\ BatchNorm \& ReLU};
\node[io, below=6mm of vstem] (vin) {Video frames\\$8\times224\times224\times3$};

%════════ audio tower (x = 6.2 cm) ════════
\node[act] (a5) at (6.2,0) {ReLU};
\node[lin, below=\g of a5] (a4) {Linear $128\to64$};
\node[act, below=\g of a4] (a3) {Dropout $p{=}0.3$};
\node[act, below=\g of a3] (a2) {ReLU};
\node[lin, below=\g of a2] (a1) {Linear $256\to128$};
\node[norm, below=6mm of a1] (astd) {Standardise};
\node[pool, below=\g of astd] (astat) {Mean \& std over time\\(log-mel and $\Delta$: $4{\times}64$)};
\node[pool, below=6mm of astat] (adel) {Temporal difference $\Delta$};
\node[conv, below=\g of adel] (amel) {Mel filterbank (64) \& log};
\node[conv, below=\g of amel] (astft) {STFT (25\,ms, hop 10\,ms)\\power spectrum};
\node[io, below=6mm of astft] (ain) {Audio waveform\\16\,kHz mono};

%════════ containers ════════
\begin{scope}[on background layer]
  \node[box, fit=(v1)(v5)] (Ev) {};
  \node[box, fit=(a1)(a5)] (Ea) {};
  \node[box, fit=(vstem)(vgap)(rc1)(radd), fill=gray!4, dashed, inner xsep=9mm, inner ysep=4mm] (rn) {};
  \node[box, fit=(astft)(astat), fill=gray!4, dashed] (lm) {};
  \node[box, fit=(rc1)(radd), fill=gray!16] (res) {};
\end{scope}
\node[side, left=2mm of Ev] {video\\encoder $E_v$\\[2pt]\footnotesize 147,776\\parameters};
\node[side, right=2mm of Ea] {audio\\encoder $E_a$\\[2pt]\footnotesize 41,152\\parameters};
\node[side, anchor=east] at (rn.west |- res) {$2\times$\\[3pt]per stage\\[3pt]$4$ stages,\\$C=64,128,$\\$256,512$};
\node[side, left=1mm of rn.south west, anchor=south east, text width=20mm] {ResNet-18\\ImageNet\\(frozen),\\per frame};
\node[side, right=2mm of lm, text width=18mm] {log-mel\\statistics\\(fixed)};

%════════ arrows inside towers ════════
\foreach \a/\b in {vin/vstem, vstem/vmp, vmp/rc1, rc1/rbn1, rbn1/rc2, rc2/rbn2, rbn2/radd, radd/vgap, vgap/vmean,
                   vmean/vstd, vstd/v1, v1/v2, v2/v3, v3/v4, v4/v5,

%════════ heads (top) ════════

\node[soft, above=\g of hv, minimum width=28mm] (sv) {Softmax};
\node[lin, above=8mm of a5, minimum width=28mm] (ha) {Linear $64\to K$};
\node[soft, above=\g of ha, minimum width=28mm] (sa) {Softmax};
\node[norm, minimum width=28mm] (cat) at ($(v5)!0.5!(a5)+(0,14.5mm)$) {Concat ($128$)};

\node[soft, above=\g of fl, minimum width=28mm] (fs) {Softmax};

\draw[up] (a5) -- node[font=\scriptsize, right] {$z_a$} (ha);
\draw[up] (hv) -- (sv); \draw[up] (ha) -- (sa); \draw[up] (cat) -- (fl); \draw[up] (fl) -- (fs);

\draw[up, rounded corners=4pt] ($(a5.north)+(0,3mm)$) -| ($(cat.south)+(6mm,0)$);

\node[io, above=4mm of sa] (oa) {audio head\\$\to\mathcal{L}_a$};
